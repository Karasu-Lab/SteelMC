1//! Terrain generation without a server: one world, no players, no tick loop.
2//!
3//! The chunks come out of the same generation pyramid the server runs
4//! (`Empty` through `Full`), so a fixed seed gives the same terrain, and each
5//! one can be read back as the packet payload a client would receive. This is
6//! what runs on wasm32, where there are no sockets to talk to a server over.
7

31/// The statuses a chunk passes through on its way to `Full`, in order.
32const PIPELINE_STATUSES: [ChunkStatus; 10] = [
33    ChunkStatus::Empty,
34    ChunkStatus::StructureStarts,
35    ChunkStatus::StructureReferences,
36    ChunkStatus::Biomes,
37    ChunkStatus::Terrain,
38    ChunkStatus::Features,
39    ChunkStatus::InitializeLight,
40    ChunkStatus::Light,
41    ChunkStatus::Spawn,
42    ChunkStatus::Full,
43];

45/// A world that generates chunks on demand and never ticks.
46pub struct HeadlessWorld {
47    runtime: Arc<Runtime>,
48    generation_pool: Arc<rayon::ThreadPool>,
49    world: Arc<World>,
50    dimension: &'static DimensionType,
51    // Keeps the damage history the world holds a weak reference to alive.
52    _damage_history: Arc<DamageHistory>,
53}
54
55impl HeadlessWorld {
56    /// Opens `generator` (for example `minecraft:overworld`) with `seed`, using
57    /// `threads` generation workers (`None` for rayon's default).
58    ///
59    /// # Errors
60    /// Returns an error if the generator is unknown or the world cannot be built.
61    pub fn open(
62        generator_key: &Identifier,
63        seed: i64,
64        threads: Option<usize>,
65    ) -> Result<Self, String> {
66        init_globals();
67
68        let runtime = Arc::new(
69            RuntimeBuilder::new_current_thread()
70                .build()
71                .map_err(|error| format!("failed to build the chunk runtime: {error}"))?,
72        );
73        let generation_pool = Arc::new({
74            let mut builder = rayon::ThreadPoolBuilder::new()
75                .thread_name(|index| format!("headless-gen-{index}"));
76            if let Some(threads) = threads {
77                builder = builder.num_threads(threads);
78            }
79            if cfg!(debug_assertions) {
80                builder = builder.stack_size(threading::DEBUG_STACK_SIZE);
81            }
82            threading::build_thread_pool(builder)
83                .map_err(|error| format!("failed to build the generation pool: {error}"))?
84        });
85
86        let registry = WorldGeneratorRegistry::new_with_builtins()?;
87        let config = registry.validate_config(generator_key, &toml::Value::Table(Map::new()))?;
88        let output = registry.create(None, &config, seed, Arc::clone(&generation_pool))?;
89        let dimension = output.dimension_type;
90        let generator = Arc::new(output.generator);
91        let generation_settings = WorldGenerationSettings::from_generator_config(
92            generator_key.clone(),
93            &output.config,
94            dimension.key.clone(),
95            dimension.min_y,
96            dimension.height,
97        );
98        let damage_history = Arc::new(DamageHistory::default());
99        let world_config = WorldConfig {
100            damage_history: Arc::clone(&damage_history),
101            game_time_source: GameTimeSource::Primary,
102            storage: WorldStorageConfig::RamOnly,
103            level_data_path: None,
104            generator,
105            generation_settings,
106            view_distance: 10,
107            simulation_distance: 10,
108            max_chained_neighbor_updates: 1_000_000,
109            compression: None,
110            is_flat: false,
111            sea_level: output.sea_level,
112            default_gamemode: GameType::Spectator,
113            difficulty: Difficulty::Peaceful,
114        };
115        let world_key = Identifier::new("headless", generator_key.path.to_string());
116        let world = runtime
117            .block_on(World::new_with_config(
118                Arc::clone(&runtime),
119                world_key,
120                dimension,
121                seed,
122                world_config,
123                Arc::clone(&generation_pool),
124            ))
125            .map_err(|error: io::Error| format!("failed to build the world: {error}"))?;
126
127        Ok(Self {
128            runtime,
129            generation_pool,
130            world,
131            dimension,
132            _damage_history: damage_history,
133        })
134    }

154    /// Generates `pos` (and the neighbors the pipeline needs) up to `Full`.
155    ///
156    /// Blocks until done; call it from a worker, never a UI thread.
157    ///
158    /// # Errors
159    /// Returns an error if a pipeline stage does not complete.
160    pub fn generate(&self, pos: ChunkPos) -> Result<(), String> {
161        let chunk_map = &self.world.chunk_map;
162        let target_step = GENERATION_PYRAMID.get_step_to(ChunkStatus::Full);
163        let radius = target_step.get_accumulated_radius_of(ChunkStatus::Empty) as i32;
164        let (min_y, height) = (self.dimension.min_y, self.dimension.height);
165
166        let cache = Arc::new(StaticCache2D::create(pos.0.x, pos.0.y, radius, {
167            let chunk_map = Arc::clone(chunk_map);
168            move |x, z| holder_at(&chunk_map, ChunkPos::new(x, z), min_y, height)
169        }));
170
171        for status in PIPELINE_STATUSES {
172            let stage_radius = target_step.get_accumulated_radius_of(status) as i32;
173            let step = GENERATION_PYRAMID.get_step_to(status);
174            let holders: Vec<Arc<ChunkHolder>> = (pos.0.y - stage_radius..=pos.0.y + stage_radius)
175                .flat_map(|z| {
176                    (pos.0.x - stage_radius..=pos.0.x + stage_radius).map(move |x| (x, z))
177                })
178                .map(|(x, z)| Arc::clone(cache.get(x, z)))
179                .collect();
180
181            // One holder at a time, in a fixed order: a stage such as `Features`
182            // writes into its neighbors, which the server's scheduler keeps
183            // apart, and running them together would make the result depend on
184            // timing.
185            let completed = self.runtime.block_on(async {
186                for holder in &holders {
187                    let Some(task) = holder.apply_step(
188                        step,
189                        chunk_map,
190                        &cache,
191                        Arc::clone(&self.generation_pool),
192                    ) else {
193                        continue;
194                    };
195                    if task.await.is_none() {
196                        return false;
197                    }
198                }
199                true
200            });
201            if !completed {
202                return Err(format!(
203                    "chunk stage {status:?} did not complete at {pos:?}"
204                ));
205            }
206        }
207        Ok(())
208    }

210    /// Generates `pos` and returns the body of the level-chunk packet a client
211    /// would be sent for it: position, section data, heightmaps and light.
212    ///
213    /// # Errors
214    /// Returns an error if generation fails or the packet cannot be written.
215    pub fn chunk_packet(&self, pos: ChunkPos) -> Result<Vec<u8>, String> {
216        self.generate(pos)?;
217        let holder = self
218            .world
219            .chunk_map
220            .tracked_chunk_holder(pos)
221            .ok_or_else(|| format!("no holder for {pos:?} after generation"))?;
222        let chunk = holder
223            .try_full_chunk()
224            .ok_or_else(|| format!("{pos:?} is not a full chunk"))?;
225        let packet = CLevelChunkWithLight {
226            x: pos.0.x,
227            z: pos.0.y,
228            chunk_data: chunk.extract_chunk_data(),
229            light_data: chunk.extract_light_data(self.dimension.has_skylight),
230        };
231        let mut out = Vec::new();
232        packet
233            .write(&mut out)
234            .map_err(|error| format!("failed to write the chunk packet: {error}"))?;
235        Ok(out)
236    }
237}

239impl HeadlessWorld {
240    /// Generates `pos` and returns every block state in it, indexed x fastest,
241    /// then z, then y upwards from [`Self::min_y`].
242    ///
243    /// # Errors
244    /// Returns an error if generation fails.
245    pub fn block_states(&self, pos: ChunkPos) -> Result<Vec<BlockStateId>, String> {
246        self.generate(pos)?;
247        let holder = self
248            .world
249            .chunk_map
250            .tracked_chunk_holder(pos)
251            .ok_or_else(|| format!("no holder for {pos:?} after generation"))?;
252        let chunk = holder
253            .try_full_chunk()
254            .ok_or_else(|| format!("{pos:?} is not a full chunk"))?;
255        let (min_y, height) = (self.dimension.min_y, self.dimension.height);
256        let mut states = Vec::with_capacity(16 * 16 * height as usize);
257        for y in min_y..min_y + height {
258            for z in 0..16 {
259                for x in 0..16 {
260                    states.push(chunk.get_block_state(BlockPos::new(
261                        pos.0.x * 16 + x,
262                        y,
263                        pos.0.y * 16 + z,
264                    )));
265                }
266            }
267        }
268        Ok(states)
269    }
270}

272/// The holder the map already tracks for `pos`, or a fresh detached one.
273fn holder_at(
274    chunk_map: &Arc<ChunkMap>,
275    pos: ChunkPos,
276    min_y: i32,
277    height: i32,
278) -> Arc<ChunkHolder> {
279    if let Some(holder) = chunk_map.tracked_chunk_holder(pos) {
280        return holder;
281    }
282    let holder = Arc::new(ChunkHolder::new(
283        pos,
284        ChunkTicketLevel::STRONGEST,
285        None,
286        min_y,
287        height,
288    ));
289    chunk_map.insert_detached_chunk_holder(pos, Arc::clone(&holder));
290    holder
291}
